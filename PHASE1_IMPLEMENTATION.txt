=== USAGE EXAMPLE ===

// 1. Initialize in App.svelte
import { logger, setupEventLogging } from '@lib/logger';
import { setupSolveHandlers, setupSessionHandlers, setupSystemHandlers } from '@events/handlers';

onMount(() => {
  logger.setConfig({
    level: isDev ? 'debug' : 'info',
    console: { enabled: true, colors: true },
  });
  
  setupEventLogging();
  setupSolveHandlers();
  setupSessionHandlers();
  setupSystemHandlers();
});

// 2. In use case
import { IEventDispatcher } from "@ports/IEventDispatcher";
import { SolveAdded } from "@events/domain";
import { logger } from "@lib/logger";

export class AddSolve {
  constructor(
    private solveRepo: ISolveRepository,
    private dispatcher: IEventDispatcher
  ) {}

  async execute(solve: Partial<Solve>): Promise<Solve> {
    logger.debug('usecase', 'AddSolve executing', { time: solve.time });
    
    const added = await this.solveRepo.addSolve(new Solve(solve));
    logger.info('usecase', 'Solve added', { solveId: added._id });
    
    await this.dispatcher.dispatch(new SolveAdded(added));
    return added;
  }
}

// 3. Subscribe to events
import { eventBus } from "@events";
import { SolveAdded } from "@events/domain";

eventBus.subscribe(SolveAdded, (event) => {
  console.log('Solve added:', event.solve);
  // Update UI, analytics, etc
});

=== KEY DESIGN DECISIONS ===

1. EventBus instead of multiple Emitters
   - Single source of truth for all events
   - Prevents circular references
   - Easier to test and debug

2. Type-Safe Events
   - No string-based event names
   - Discriminated unions possible
   - IDE autocomplete support

3. Priority-Based Handlers
   - Higher priority handlers execute first
   - Useful for cascading updates
   - One-time subscribers supported

4. Async All The Way
   - All handlers are awaited
   - Sequential execution
   - Error handling per handler

5. Injection Over Singletons
   - Use cases receive IEventDispatcher
   - Easier to mock in tests
   - IPC adapters can inject also
