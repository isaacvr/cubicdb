// src/lib/core/usecases/AddAlgorithm.ts
import type { IAlgorithmRepository } from "../ports/IAlgorithmRepository";
import type { IEventDispatcher } from "../ports/IEventDispatcher";
import type { Algorithm } from "@interfaces";
import { AlgorithmAdded } from "@events/domain";

export class AddAlgorithm {
  constructor(
    private repo: IAlgorithmRepository,
    private dispatcher: IEventDispatcher
  ) {}

  async execute(algorithm: Algorithm): Promise<Algorithm> {
    // Validate
    if (!algorithm.shortName) throw new Error("Algorithm must have shortName");
    
    // Persist
    const added = await this.repo.addAlgorithm(algorithm);
    
    // Notify
    await this.dispatcher.dispatch(new AlgorithmAdded(added));
    
    return added;
  }
}

=== TEMPLATE 6: Subscribe in Component ===

// src/pages/Your/YourPage.svelte
<script lang="ts">
  import { onMount, onDestroy } from 'svelte';
  import { eventBus } from '@events';
  import { YourEntityCreated, YourEntityUpdated } from '@events/domain';
  import type { EventSubscription } from '@events/types';

  let entities = [];
  let subscriptions: EventSubscription[] = [];

  onMount(() => {
    // Subscribe to events
    subscriptions.push(
      eventBus.subscribe(YourEntityCreated, (event) => {
        entities = [...entities, event.entity];
      })
    );

    subscriptions.push(
      eventBus.subscribe(YourEntityUpdated, (event) => {
        entities = entities.map(e => 
          e.id === event.updatedEntity.id ? event.updatedEntity : e
        );
      })
    );
  });

  onDestroy(() => {
    subscriptions.forEach(sub => sub.unsubscribe());
  });
</script>

=== EVENT NAMING CONVENTIONS ===

Pattern: [Entity][Action]
- EntityCreated (Create operation)
- EntityUpdated (Update operation)
- EntityDeleted (Delete operation)
- EntitySelected (Selection change)
- EntityStatusChanged (Status-specific update)
- EntitiesRemoved (Bulk delete)

Examples:
✅ SolveAdded, SessionUpdated, AlgorithmRemoved
❌ OnSolveAdd, SolveAddedEvent, CreateSolve (too verbose)

=== COMMON PATTERNS ===

Pattern A: Optimistic Update
const optimisticEntity = { ...entity, status: 'pending' };
const result = await useCase.execute(entity);
if (!result) revertOptimistic();

Pattern B: Bulk Operations
const removed = await useCase.removeMany([e1, e2, e3]);
dispatch(new EntitiesRemoved(removed));

Pattern C: Before/After Events
dispatch(new EntityUpdated(previousState, newState));
// Allows calculating deltas in handlers

Pattern D: Sequential Processing
eventBus.subscribe(EntityCreated, async (e) => {
  await updateRelated(e);
  dispatch(new EntityRelatedUpdated(related));
});

=== CIRCULAR REFERENCE PREVENTION ===

❌ DON'T DO THIS:
EventA → triggers handler → dispatch EventB
EventB → triggers handler → dispatch EventA
// EventBus will warn and prevent re-entry

✅ DO THIS:
EventA → handler updates state
State change → separate observer triggers EventB
// Use separate event for the cascade
